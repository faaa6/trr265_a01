function fit_loop(models, formulas, data, contr)
   df = DataFrame()
   for i=1:length(models)
   global m = fit(MixedModel, formulas[i], data, contrasts=contr);
   df = append!(df,DataFrame(models = models[i], deviance = MixedModels.deviance(m), singular = issingular(m)));
   @eval $(Symbol(models[i])) = deepcopy(m);
   end
   return df
end

function glmm_fit_loop(models, formulas, data, contr)
   df = DataFrame()
   for i=1:length(models)
   global m = fit(MixedModel, formulas[i], data, dist; nAGQ, contrasts=contr);
   df = append!(df,DataFrame(models = models[i], singular = issingular(m)));
   @eval $(Symbol(models[i])) = deepcopy(m);
   end
   return df
end

function row_diff(a)
   b = similar(a)
   b[1,:] .= 0
   for i in 2:size(a,1)
         b[i,:] .= a[i,:] .- a[i-1,:] 
   end
   return b 
end

function lrt_table(nms,mods,lrt)
   DataFrame(;
      name = nms, 
      dof=MixedModels.dof.(mods),
      dof_dif=row_diff(MixedModels.dof.(mods)),
      deviance=MixedModels.deviance.(mods),
      deviance_diff=row_diff(MixedModels.deviance.(mods)),
      AIC=aic.(mods),
      AIC_dif=row_diff(aic.(mods)),
      BIC=bic.(mods),
      BIC_dif=row_diff(bic.(mods)),
      pvalue=vcat(:., last(round.(lrt.lrt.pval, digits=3))),
      ) 
end

1.1 Problem

There is variation associated within the time of year. Fitting this variation in a linear model using passing time as a continuous variable poses two problems:

  1. linearity: Within one year, assumption of linearity accross proceding time is violated, as the highest valus (last day e.g. 31st of December) and lowest value (first day e.g. 1st of January) are back to back
  2. centering: Within one year, centering is arbitrary but can affect the model fit.

1.2 Solution

To avoid the assumption of linearity the proceding in days will be transformde over one itteration of a sinus curve (i.e. 0 - 2π).

To avoid centering problem, the onset of the sinus curve will be moved along each day within one year, with models being generated for each proceding day.

This will be done using the following formula:

\[ sin(\frac{"c" + 0:365}{365*2π}) \]

with c being every number between 0 and 365

Code 
fig = Figure()
ax1 = Axis(fig[1, 1], title = "model", xlabel = "YearDay", ylabel = "sin(YearDay/(365*2π))")
plot!(ax1, 0:365, sin.((0:365)./365*2π))

Mathematically, this apporach further can be justified by the tilt of the earth axis and the rotational exposure to the sun, which also follows an sinus curve.

1.3.2 model fit loop

Now we fit procedual models for every day and collect the model improvement on the m0_tepm model for each day-variable

Code 
ftemp_sin = [@formula temp ~ 1 + yd_sin + (1 | center);]
mtemp_sin = ["mtemp_"]
df_temp = DataFrame()
for i in 0:365
  meteo_sin = @chain meteo begin
    @transform(:yd_sin = sin((i .+ :day)./365 .*2π))
    end
  fit_loop(mtemp_sin, ftemp_sin, meteo_sin, contr_temp)
  @eval $(Symbol(string(mtemp_sin[1], i))) = deepcopy(m);
  df_temp = append!(df_temp,DataFrame(model = string(mtemp_sin[1], i), I = i, deviance = MixedModels.deviance(m), AIC = aic(m), BIC = bic(m), singular = issingular(m)));
  end

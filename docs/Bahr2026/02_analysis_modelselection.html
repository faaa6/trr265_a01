function row_diff(a)
   b = similar(a)
   b[1,:] .= 0
   for i in 2:size(a,1)
         b[i,:] .= a[i,:] .- a[i-1,:] 
   end
   return b 
end

function lrt_table(nms,mods,lrt)
   DataFrame(;
      name = nms, 
      dof=MixedModels.dof.(mods),
      dof_dif=row_diff(MixedModels.dof.(mods)),
      deviance=MixedModels.deviance.(mods),
      deviance_diff=row_diff(MixedModels.deviance.(mods)),
      AIC=aic.(mods),
      AIC_dif=row_diff(aic.(mods)),
      BIC=bic.(mods),
      BIC_dif=row_diff(bic.(mods)),
      pvalue=round.(lrt.lrt.pval, digits=3),
      ) 
end

function fit_loop(models, formulas, data)
   df = DataFrame()
   for i=1:length(models)
   global m = fit(MixedModel, formulas[i], data, contrasts=contr);
   df = append!(df,DataFrame(models = models[i], singular = issingular(m)));
   @eval $(Symbol(models[i])) = deepcopy(m);
   end
   return df
end

function fit_loop_save(models, formulas, data, location)
   df = DataFrame()
   for i=1:length(models)
   global m = fit(MixedModel, formulas[i], data; contrasts=contr);
   saveoptsum(location * models[i] * ".json", m);
   df = append!(df,DataFrame(models = models[i], singular = issingular(m)));
   @eval $(Symbol(models[i])) = deepcopy(m);
   end
   return df
end

function fit_loop_load(models, formulas, data, location)
   df = DataFrame()
   for i=1:length(models)
   global m = LinearMixedModel(formulas[i], data; contrasts=contr);
   restoreoptsum!(m, location * models[i] * ".json");
   df = append!(df,DataFrame(models = models[i], singular = issingular(m)));
   @eval $(Symbol(models[i])) = deepcopy(m);
   end
   return df
end

function glmm_fit_loop(models, formulas, data)

   df = append!(df,DataFrame(models = models[i], singular = issingular(m)));
   @eval $(Symbol(models[i])) = deepcopy(m);
   end
   return df
end

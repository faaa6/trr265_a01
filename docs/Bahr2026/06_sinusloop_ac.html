function fit_loop(models, formulas, data, contr)
   df = DataFrame()
   for i=1:length(models)
   global m = fit(MixedModel, formulas[i], data, contrasts=contr);
   df = append!(df,DataFrame(models = models[i], deviance = MixedModels.deviance(m), singular = issingular(m)));
   @eval $(Symbol(models[i])) = deepcopy(m);
   end
   return df
end

function glmm_fit_loop(models, formulas, data, contr)
   df = DataFrame()
   for i=1:length(models)
   global m = fit(MixedModel, formulas[i], data, dist; nAGQ, contrasts=contr);
   df = append!(df,DataFrame(models = models[i], singular = issingular(m)));
   @eval $(Symbol(models[i])) = deepcopy(m);
   end
   return df
end

function row_diff(a)
   b = similar(a)
   b[1,:] .= 0
   for i in 2:size(a,1)
         b[i,:] .= a[i,:] .- a[i-1,:] 
   end
   return b 
end

function lrt_table(nms,mods,lrt)
   DataFrame(;
      name = nms, 
      dof=MixedModels.dof.(mods),
      dof_dif=row_diff(MixedModels.dof.(mods)),
      deviance=MixedModels.deviance.(mods),
      deviance_diff=row_diff(MixedModels.deviance.(mods)),
      AIC=aic.(mods),
      AIC_dif=row_diff(aic.(mods)),
      BIC=bic.(mods),
      BIC_dif=row_diff(bic.(mods)),
      pvalue=vcat(:., last(round.(lrt.lrt.pval, digits=3))),
      ) 
end

1.1 Model approach

First we fit simple models accounting for variance associated specifically with

  • wd ~ weekday-effect
  • hd ~ holiday-effect
  • jd ~ dry Januray/drinking December effect
  • ld ~ lockdown effect
  • a1 ~ age/birth cohort effect
  • gender ~ gender effect

for each measure of AC:

  • overall AC [LMM]
  • AC frequency [GLMM]
  • AC amounts [LMM]

Then, following the approach from above, 366 models for each AC measure are fit with a sinus transformed yearday variable that is centered at each day in the year. Out of these approach the two models with the best fit are being selected and their model improvment is being assessed.

df_ac = DataFrame()

2 Summary

All AC models yield a similar pattern

  • overall AC: centerd at day 248
  • drinking frequency: centerd at 248
  • drinking amounts: centered at 250

Which models quite closely to maximum outsiode temperature

  • maximum outside temperature: centered at 258

While the maximum outside temperature follows a naturally occurring sinus pattern, the results indicate that the modelled approache fits this association quite well.

Transfering that approach into data that does not directly bare the association with a naturally occurring sinus pattern, the approach still yields interpretable and useful information.

It would be intersting to see, how the model would behave if daily temperature would be implemented as a predictor into the fiyed effect structure.
